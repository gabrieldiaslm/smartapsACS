# 3. CRIANDO CRIANÇAS
nomes_meninos = ['Miguel', 'Arthur', 'Gael','Geralt', 'Heitor', 'Theo', 'Davi', 'Bernardo', 'Gabriel', 'Lemo', 'Cabra']
nomes_meninas = ['Helena', 'Alice', 'Laura', 'Noele', 'Maria Alice', 'Sophia', 'Manuela', 'Lia', 'Emilia', 'Emmy']
sobrenomes = ['Silva', 'Santos', 'Oliveira', 'Souza', 'Rodrigues', 'Ferreira', 'Lima']
localidades = ['Centro', 'Bairro São José', 'Nova Parnaíba', 'Rodoviária', 'Reis Veloso']

def gerar_cns_falso():
    return str(random.randint(100000000000000, 999999999999999))

hoje = date.today()

for i in range(120):
    sexo = random.choice(['M', 'F'])
    
    primeiro_nome = random.choice(nomes_meninos) if sexo == 'M' else random.choice(nomes_meninas)
    nome = f"{primeiro_nome} {random.choice(sobrenomes)} {random.choice(sobrenomes)}"
    nome_mae = f"{random.choice(nomes_meninas)} {random.choice(sobrenomes)} {random.choice(sobrenomes)}"
    
    # Gera idades entre 10 dias e 9 anos
    dias_idade = random.randint(10, 3200)
    data_nascimento = hoje - timedelta(days=dias_idade)
    
    cns = gerar_cns_falso()
    
    if not Crianca.objects.filter(cns=cns).exists():
        nova_crianca = Crianca.objects.create(
            nome=nome,
            data_nascimento=data_nascimento,
            cns=cns,
            localidade=random.choice(localidades),
            nome_mae=nome_mae,
            sexo=sexo
        )
        # O Signal tratará de criar o cartão!
        # Agora forçamos o "verificar_atrasos" para pintar o painel de vermelho nas vacinas vencidas
        nova_crianca.verificar_atrasos()

print("✅ 30 Crianças cadastradas com sucesso!")
